# The following lines of boilerplate have to be in your project's
# CMakeLists in this exact order for cmake to work correctly
cmake_minimum_required(VERSION 3.16)

# the part every kibbl. firmware shares (kibbl_core)
set(EXTRA_COMPONENT_DIRS "../components")

# BUILD_INFO.h (create_build_info.sh in CI, a WIP version locally) is also the version
# in the image, which the server shows when an update is uploaded. VERSION holds the
# version number: raise it with every change to the firmware
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/VERSION" fw_number)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/VERSION")
set(build_info "")
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/main/BUILD_INFO.h")
    file(READ "${CMAKE_CURRENT_LIST_DIR}/main/BUILD_INFO.h" build_info)
endif()
# a local one follows VERSION; one of create_build_info.sh stays as it is
if(build_info STREQUAL "" OR build_info MATCHES "generated by CMakeLists.txt")
    file(WRITE "${CMAKE_CURRENT_LIST_DIR}/main/BUILD_INFO.h"
        "// generated by CMakeLists.txt for a local build\n"
        "#ifndef BUILD_INFO_H\n#define BUILD_INFO_H\n#define FW_VERSION \"KIBBL-${fw_number}-WIP\"\n#endif\n")
endif()
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/main/BUILD_INFO.h" fw_version_line REGEX "#define FW_VERSION")
string(REGEX REPLACE ".*\"(.*)\".*" "\\1" PROJECT_VER "${fw_version_line}")

include($ENV{IDF_PATH}/tools/cmake/project.cmake)

project(KIBBL-FW)
